  </div>

  <div class="button-group">
    <button class="button confirm-btn" onclick="confirmSubmit()">Print</button>
  <button class="button edit-btn" onclick="goBack()">ยืนยัน</button>
  </div>

  <script>
    function confirmSubmit() {
    window.print(); // เปิดหน้าพิมพ์
  }

    function goBack() {
    window.history.back();
  }
    // โหลดข้อมูลจาก localStorage
    const data = JSON.parse(localStorage.getItem('enrollData'));
    const infoBox = document.getElementById('infoBox');

    if (!data) {
      infoBox.innerHTML = '<p>ไม่พบข้อมูลใบสมัคร กรุณากรอกฟอร์มใหม่</p>';
    } else {
      infoBox.innerHTML = `
        <p><strong>ชื่อ - นามสกุล:</strong> ${data.name}</p>
        <p><strong>เลขประจำตัวประชาชน:</strong> ${data.id}</p>
        <p><strong>เบอร์โทรศัพท์:</strong> ${data.phone}</p>
        <p><strong>หลักสูตรที่สมัคร:</strong> ${data.course}</p>
        <p><strong>ที่อยู่:</strong> ${data.address}</p>
      `;
    }

    document.getElementById('confirmBtn').addEventListener('click', () => {
      alert('ส่งข้อมูลเรียบร้อยแล้ว ขอบคุณค่ะ');
      localStorage.removeItem('enrollData');
      window.location.href = 'index.html';
    });

    document.getElementById('editBtn').addEventListener('click', () => {
      window.location.href = 'index.html';
    });
  </script>
