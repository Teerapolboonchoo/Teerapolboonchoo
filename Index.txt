  <script>
    let currentFormType = null;

    function showForm() {
      const value = document.getElementById("formSelector").value;
      document.getElementById("enrollForm").style.display = "none";
      document.getElementById("withdrawForm").style.display = "none";
      document.getElementById("resignForm").style.display = "none";

      if (value === "enroll") document.getElementById("enrollForm").style.display = "block";
      else if (value === "withdraw") document.getElementById("withdrawForm").style.display = "block";
      else if (value === "resign") document.getElementById("resignForm").style.display = "block";
    }

    function openPopup(formType) {
      currentFormType = formType;

      if (formType === "enroll") {
        const data = {
          name: document.getElementById("enrollName").value,
          id: document.getElementById("enrollId").value,
          phone: document.getElementById("enrollPhone").value,
          course: document.getElementById("enrollCourse").value,
          address: document.getElementById("enrollAddress").value
        };
        localStorage.setItem("enrollData", JSON.stringify(data));
      } else if (formType === "withdraw") {
        const data = {
          name: document.getElementById("withdrawName").value,
          amount: document.getElementById("withdrawAmount").value,
          reason: document.getElementById("withdrawReason").value
        };
        localStorage.setItem("withdrawData", JSON.stringify(data));
      }

      document.getElementById("popup").style.display = "flex";
    }

    function confirmForm(confirm) {
      document.getElementById("popup").style.display = "none";
      if (confirm) {
        if (currentFormType === "enroll") {
          window.location.href = "confirm.html";
        } else if (currentFormType === "withdraw") {
          window.location.href = "confirm-withdraw.html";
        }
      }
    }
  </script>
