  <!-- JavaScript Section -->
  <script>
    function showForm() {
      const value = document.getElementById("formSelector").value;
      document.getElementById("enrollForm").style.display = "none";
      document.getElementById("withdrawForm").style.display = "none";
      document.getElementById("resignForm").style.display = "none";
      if (value === "enroll") document.getElementById("enrollForm").style.display = "block";
      else if (value === "withdraw") document.getElementById("withdrawForm").style.display = "block";
      else if (value === "resign") document.getElementById("resignForm").style.display = "block";
    }

    function submitForm(event) {
      event.preventDefault();
      document.getElementById("popup").style.display = "flex";
    }

    function confirmForm(confirm) {
      document.getElementById("popup").style.display = "none";
      if (confirm) {
        const form = document.getElementById("enrollRealForm");
        const newData = {
          name: form.name.value,
          id: form.id.value,
          phone: form.phone.value,
          course: form.course.value,
          address: form.address.value
        };

        let enrollData = JSON.parse(localStorage.getItem("enrollData") || "[]");
        enrollData.push(newData);
        localStorage.setItem("enrollData", JSON.stringify(enrollData));

        window.location.href = "confirm.html";
        
        function openPopup() {
  const data = {
    name: document.querySelector("#enrollForm input:nth-of-type(1)").value,
    id: document.querySelector("#enrollForm input:nth-of-type(2)").value,
    phone: document.querySelector("#enrollForm input:nth-of-type(3)").value,
    course: document.querySelector("#enrollForm input:nth-of-type(4)").value,
    address: document.querySelector("#enrollForm textarea").value,
    gpa: document.querySelector("#gpa").value,
    major: document.querySelector("#major").value,
    prevSchool: document.querySelector("#prevSchool").value
  };

  const existingData = JSON.parse(localStorage.getItem("enrollData") || "[]");
  existingData.push(data);
  localStorage.setItem("enrollData", JSON.stringify(existingData));

  document.getElementById("popup").style.display = "flex";
}
      }
    }
  </script>
